    if (!this.ctx) return;
    try {
      if (this.ctx.state === 'suspended') this.ctx.resume();
      const osc = this.ctx.createOscillator();
      const gain = this.ctx.createGain();
      osc.type = type;
      osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
      gain.gain.setValueAtTime(0.08, this.ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);
      osc.connect(gain);
      gain.connect(this.ctx.destination);
      osc.start();
      osc.stop(this.ctx.currentTime + duration);
    } catch(e){}
  }
  click() { this.play(800, 'triangle', 0.04); }
  hit() { this.play(150, 'sawtooth', 0.12); }
  loot() { this.play(950, 'sine', 0.15); }
  jackpot() {
    [523.25, 659.25, 783.99, 1046.50].forEach((f, idx) => {
      setTimeout(() => this.play(f, 'square', 0.25), idx * 90);
    });
  }

class Game {
  constructor() {
    this.audio = new SoundFx();
    this.state = this.defaultState();
    this.invFilter = 'all';
    this.shopMode = 'buy';
    this.combat = null;
    this.isBusy = false;

    this.inv = new InvManager(this);
    this.ui = new UIEngine(this);
    this.log = new LogManager(this);
    this.trader = new TraderManager(this);
    this.gram = new GramManager(this);

    this.load();
    this.initTimer();
  }

  defaultState() {
    return {
      name: "Стрелок",
      nameCreated: false,
      title: "Новичок",
      level: 1,
      exp: 0,
      expNext: 100,
      hp: 100,
      maxHp: 100,
      energy: 100,
      maxEnergy: 100,
      radiation: 0,
      // Hunger & Thirst (0 to 100)
      hunger: 100,
      thirst: 100,
      // Attributes (Stats) & Free Points
      stats: { str: 5, agi: 5, end: 5, per: 5 },
      statPoints: 5,
      rubles: 450,
      gramcoin: 0.00,
      equipment: { weapon: "knife_rusty", armor: "armor_jacket", artifact: null },
      inventory: [
        { id: "item_bread", count: 2 },
        { id: "item_water", count: 3 },
        { id: "med_bandage", count: 2 },
        { id: "item_food", count: 1 }
      ],
      quests: {},
      achievements: {},
      tracker: { kills: 0, searches: 0, gramTotal: 0, radHeals: 0, kordon: 0, swamp: 0, forest: 0, factory: 0, valley: 0, bunker: 0, lab18: 0, jupiter: 0, radar: 0, tunnel: 0, sarc: 0 }
    };
  }

  save() {
    try {
      localStorage.setItem("zona_z_save_v4", JSON.stringify(this.state));
    } catch(e){}
  }

  load() {
    try {
      let d = localStorage.getItem("zona_z_save_v4");
      if (!d) d = localStorage.getItem("zona_z_save_v3");
      if (d) {
        const parsed = JSON.parse(d);
        this.state = Object.assign(this.defaultState(), parsed);
        if (!this.state.stats) this.state.stats = { str: 5, agi: 5, end: 5, per: 5 };
        if (this.state.statPoints === undefined) this.state.statPoints = 5;
        if (this.state.hunger === undefined) this.state.hunger = 100;
        if (this.state.thirst === undefined) this.state.thirst = 100;
        if (!this.state.equipment) this.state.equipment = { weapon: "knife_rusty", armor: "armor_jacket", artifact: null };
        if (!this.state.tracker) this.state.tracker = { kills: 0, searches: 0, gramTotal: 0, radHeals: 0 };
        
        // Normalize legacy boolean achievements
        if (this.state.achievements) {
          for (const k in this.state.achievements) {
            if (typeof this.state.achievements[k] === 'boolean') {
              this.state.achievements[k] = { unlocked: this.state.achievements[k], claimed: false };
            }
          }
        }
      }
    } catch(e){}
    this.recalc();
    this.ui.renderAll();
    this.checkQ();
    this.checkAch();

    if (!this.state.nameCreated) {
      setTimeout(() => this.openNickModal(), 300);
    }
  }

  openNickModal() {
    const m = $('m-name');
    if (m) {
      $('inp-nick').value = this.state.name === "Стрелок" && !this.state.nameCreated ? "" : this.state.name;
      m.classList.add('active');
      setTimeout(() => { const el = $('inp-nick'); if (el && el.focus) el.focus(); }, 150);
    }
  }

  pickNick(val) {
    $('inp-nick').value = val;
    this.audio.click();
  }

  submitNick() {
    let val = ($('inp-nick').value || '').trim();
    if (!val) val = "Стрелок";
    this.state.name = val;
    this.state.nameCreated = true;
    $('m-name').classList.remove('active');
    this.audio.loot();
    this.ui.header();
    this.ui.profile();
    this.save();
    this.ui.toast(`Добро пожаловать в Зону, ${val}! Позывной сохранён.`);
  }

  // ------------------------------------------------------
  // ПРОКАЧКА ХАРАКТЕРИСТИК (ATTRIBUTES)
  // ------------------------------------------------------
  allocStat(statKey) {
    if (this.state.statPoints <= 0) {
      this.ui.toast("Нет свободных очков характеристик!", "red");
      return;
    }
    if (!this.state.stats[statKey] && this.state.stats[statKey] !== 0) return;

    this.state.statPoints--;
    this.state.stats[statKey]++;
    this.audio.loot();
    this.recalc();
    this.ui.profile();
    this.ui.bars();
    this.ui.gear();
    this.save();

    const names = { str: "Сила", agi: "Ловкость", end: "Выносливость", per: "Внимательность" };
    this.ui.toast(`Улучшено: ${names[statKey]} -> ${this.state.stats[statKey]}!`, "green");
    this.log.add(`Характеристика [${names[statKey]}] повышена до ${this.state.stats[statKey]}.`);
  }

  // ------------------------------------------------------
  // ПРОВЕРКА ПОТРЕБНОСТЕЙ: ЕСЛИ 0, ДЕЙСТВИЯ ЗАПРЕЩЕНЫ!
  // ------------------------------------------------------
  canAct(showWarn = true) {
    if (this.state.hunger <= 0) {
      if (showWarn) {
        this.audio.hit();
        this.ui.toast("💀 ВЫ ОБЕССИЛЕНЫ ОТ ГОЛОДА (0%)! Никакие действия невозможны, пока вы не поедите!", "red");
        this.ui.openNeedsModal('hunger');
      }
      return false;
    }
    if (this.state.thirst <= 0) {
      if (showWarn) {
        this.audio.hit();
        this.ui.toast("💀 ВЫ УМИРАЕТЕ ОТ ЖАЖДЫ (0%)! Никакие действия невозможны, пока вы не попьёте!", "red");
        this.ui.openNeedsModal('thirst');
      }
      return false;
    }
    return true;
  }

  recalc() {
    let hpMax = 100 + (this.state.stats.str * 4);
    let enMax = 100 + (this.state.stats.end * 3);

    ['weapon', 'armor', 'artifact'].forEach(slot => {
      const id = this.state.equipment[slot];
      if (id && DB.items[id]) {
        const it = DB.items[id];
        if (it.bonus) {
          if (it.bonus.maxHp) hpMax += it.bonus.maxHp;
          if (it.bonus.maxEnergy) enMax += it.bonus.maxEnergy;
        }
      }
    });

    this.state.maxHp = hpMax;
    this.state.maxEnergy = enMax;
    this.state.hp = Math.min(this.state.hp, this.state.maxHp);
    this.state.energy = Math.min(this.state.energy, this.state.maxEnergy);

    // Title progression
    const titles = ["Новичок", "Опытный", "Бывалый", "Ветеран", "Мастер", "Гуру", "Легенда Зоны", "Хранитель Ноосферы"];
    const idx = Math.min(titles.length - 1, Math.floor((this.state.level - 1) / 7));
    this.state.title = titles[idx];
  }

  initTimer() {
    setInterval(() => {
      let ch = false;
      // Energy regen
      if (this.state.energy < this.state.maxEnergy) {
        this.state.energy = Math.min(this.state.maxEnergy, this.state.energy + 1);
        ch = true;
      }
      // Radiation damage
      if (this.state.radiation > 0 && Math.random() < 0.25) {
        this.state.hp = Math.max(1, this.state.hp - 1);
        ch = true;
      }
      // Natural Hunger & Thirst burn (mitigated by Endurance)
      const burnMult = Math.max(0.4, 1 - (this.state.stats.end * 0.015));
      if (this.state.hunger > 0) {
        this.state.hunger = Math.max(0, Number((this.state.hunger - (0.2 * burnMult)).toFixed(1)));
        ch = true;
      }
      if (this.state.thirst > 0) {
        this.state.thirst = Math.max(0, Number((this.state.thirst - (0.3 * burnMult)).toFixed(1)));
        ch = true;
      }

      if (ch) {
        this.ui.bars();
        this.ui.updateNeedsBanner();
        this.save();
      }
    }, 4500);
  }

  restCamp() {
    this.audio.click();
    // Campfire emergency relief if player has 0 food, 0 water and 0 rubles
    if (this.state.hunger <= 0 || this.state.thirst <= 0) {
      const hasFood = this.state.inventory.some(i => DB.items[i.id] && DB.items[i.id].hunger);
      const hasDrink = this.state.inventory.some(i => DB.items[i.id] && DB.items[i.id].thirst);
      if (!hasFood || !hasDrink) {
        if (this.state.hunger <= 0) this.state.hunger = 25;
        if (this.state.thirst <= 0) this.state.thirst = 30;
        this.ui.toast("☕ Сталкеры у костра пожалели вас: дали сухарь (+25%) и глоток воды (+30%)!", "green");
        this.log.add("Добрые сталкеры у костра спасли вас от голодной смерти.");
      }
    }

    this.state.energy = Math.min(this.state.maxEnergy, this.state.energy + 35);
    this.state.hp = Math.min(this.state.maxHp, this.state.hp + 20);
    this.ui.bars();
    this.ui.updateNeedsBanner();
    this.ui.toast("☕ Вы отдохнули у сталкерского костра (+35⚡, +20 HP)");
    this.log.add("Отдых у костра восстановил силы.");
    this.save();
  }

  doctorHeal() {
    this.audio.click();
    if (this.state.rubles < 150) {
      this.ui.toast("Недостаточно денег! Врач требует 150₽", "red");
      return;
    }
    this.state.rubles -= 150;
    this.state.hp = this.state.maxHp;
    this.state.radiation = 0;
    this.state.hunger = Math.max(this.state.hunger, 40);
    this.state.thirst = Math.max(this.state.thirst, 50);
    this.state.tracker.radHeals = (this.state.tracker.radHeals || 0) + 1;
    this.ui.header();
    this.ui.bars();
    this.ui.updateNeedsBanner();
    this.ui.toast("💉 Врач полностью восстановил здоровье, снял радиацию и напоил глюкозой!", "green");
    this.log.add("Полевой медик провел полное обеззараживание (-150₽).");
    this.checkAch();
    this.save();
  }

  unequip(slot) {
    const cur = this.state.equipment[slot];
    if (!cur) return;
    this.inv.add(cur, 1);
    this.state.equipment[slot] = null;
    this.recalc();
    this.audio.click();
    this.ui.gear();
    this.ui.bars();
    this.ui.toast("Предмет снят в рюкзак");
    this.save();
  }

  setInvFilter(f, btn) {
    this.invFilter = f;
    document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
    if (btn) btn.classList.add('active');
    this.ui.inventory();
  }

  resetConfirm() {
    if (confirm("ВНИМАНИЕ! Полностью стереть прогресс и начать игру заново?")) {
      localStorage.removeItem("zona_z_save_v4");
      localStorage.removeItem("zona_z_save_v3");
      this.state = this.defaultState();
      this.save();
      this.recalc();
      this.ui.renderAll();
      this.ui.toast("Прогресс сброшен.");
      setTimeout(() => this.openNickModal(), 200);
    }
  }

  getAtk() {
    const wid = this.state.equipment.weapon;
    const w = wid ? DB.items[wid] : { dmg: [5, 8] };
    let min = w.dmg[0] + Math.floor(this.state.level * 1.5) + (this.state.stats.str * 2);
    let max = w.dmg[1] + Math.floor(this.state.level * 2) + (this.state.stats.str * 3);

    const aid = this.state.equipment.artifact;
    if (aid && DB.items[aid] && DB.items[aid].bonus && DB.items[aid].bonus.atk) {
      min += DB.items[aid].bonus.atk;
      max += DB.items[aid].bonus.atk;
    }
    const arm = this.state.equipment.armor;
    if (arm && DB.items[arm] && DB.items[arm].bonus && DB.items[arm].bonus.atk) {
      min += DB.items[arm].bonus.atk;
      max += DB.items[arm].bonus.atk;
    }
    return [min, max];
  }

  getDef() {
    let d = Math.floor(this.state.level * 1.2) + Math.floor(this.state.stats.end * 0.8);
    const aid = this.state.equipment.armor;
    if (aid && DB.items[aid]) d += (DB.items[aid].def || 0);
    const artId = this.state.equipment.artifact;
    if (artId && DB.items[artId] && DB.items[artId].bonus && DB.items[artId].bonus.def) {
      d += DB.items[artId].bonus.def;
    }
    return d;
  }

  checkLvl() {
    let leveled = false;
    while (this.state.exp >= this.state.expNext) {
      this.state.exp -= this.state.expNext;
      this.state.level++;
      this.state.statPoints += 3; // +3 Free Attribute points per level!
      this.state.expNext = Math.floor(this.state.expNext * 1.35 + 40);
      this.state.hp = this.state.maxHp;
      this.state.energy = this.state.maxEnergy;
      leveled = true;
      this.ui.toast(`🎉 НОВЫЙ УРОВЕНЬ: ${this.state.level}! Получено +3 очка характеристик!`, "gold");
      this.log.add(`Получен ${this.state.level} ранг сталкера! Доступно +3 очка характеристик.`);
    }
    if (leveled) {
      this.recalc();
      this.ui.header();
      this.ui.bars();
      this.ui.profile();
      this.ui.locations();
      this.checkQ();
      this.checkAch();
      this.save();
    }
  }

  // ------------------------------------------------------
  // ВЫЛАЗКИ И ИССЛЕДОВАНИЯ
  // ------------------------------------------------------
  explore(locId) {
    if (this.isBusy) return;
    if (this.combat) { this.ui.toast("Сначала завершите бой!", "red"); return; }
    
    // MANDATORY NEEDS CHECK: NO ACTIONS WHEN HUNGER OR THIRST IS 0!
    if (!this.canAct(true)) return;

    const loc = DB.locations.find(l => l.id === locId);
    if (!loc) return;
    if (this.state.level < loc.minLevel) { this.ui.toast(`Требуется ${loc.minLevel} уровень!`, "red"); return; }
    if (this.state.hp <= 15) { this.ui.toast("Вы тяжело ранены! Полечитесь перед вылазкой.", "red"); return; }
    if (this.state.energy < loc.energyCost) { this.ui.toast(`Мало энергии! Нужно ${loc.energyCost}⚡.`, "red"); return; }

    this.isBusy = true;
    this.state.energy -= loc.energyCost;
    this.state.tracker.searches++;

    // Consume Hunger and Thirst on exploration (endurance reduces rate)
    const drainMult = Math.max(0.4, 1 - (this.state.stats.end * 0.015));
    this.state.hunger = Math.max(0, Number((this.state.hunger - (4.0 * drainMult)).toFixed(1)));
    this.state.thirst = Math.max(0, Number((this.state.thirst - (6.0 * drainMult)).toFixed(1)));

    // Track location counts
    const key = loc.id.replace('loc_', '');
    if (this.state.tracker[key] !== undefined) this.state.tracker[key]++;

    this.audio.play(350, 'triangle', 0.1);
    this.ui.bars();
    this.ui.updateNeedsBanner();

    if (loc.minLevel >= 3) {
      const r = Math.floor(Math.random() * 8) + 2;
      this.state.radiation = Math.min(100, this.state.radiation + r);
    }

    // Boosted Gramcoin Drop Roll (enhanced by Perception stat!)
    const perBonus = (this.state.stats.per * 0.0008);
    this.gram.roll(loc.gramcoinChance + perBonus);

    // Event/Combat/Loot resolution
    setTimeout(() => {
      this.isBusy = false;
      const roll = Math.random();
      if (roll < 0.44) {
        this.startCombat(loc);
      } else if (roll < 0.76) {
        this.findLoot(loc);
      } else {
        this.triggerEvent();
      }
      this.checkQ();
      this.checkAch();
      this.save();
    }, 280);
  }

  findLoot(loc) {
    const pool = loc.loot;
    const itId = pool[Math.floor(Math.random() * pool.length)];
    const it = DB.items[itId];
    if (!it) return;

    this.inv.add(itId, 1);
    const expGained = Math.floor(loc.minLevel * 8 + 15);
    this.state.exp += expGained;
    this.audio.loot();
    
    let toastType = '';
    if (it.rarity === 'mythic') toastType = 'red';
    else if (it.rarity === 'legendary') toastType = 'gold';

    this.ui.toast(`📦 Найдено: [${it.rarityName}] ${it.name} (+${expGained} EXP)`, toastType);
    this.log.add(`Найдено: [${it.rarityName}] ${it.name} (+${expGained} EXP).`);
    this.checkLvl();
    this.ui.bars();
  }

  triggerEvent() {
    const ev = DB.events[Math.floor(Math.random() * DB.events.length)];
    const m = $('m-event');
    $('ev-title').textContent = ev.title;
    $('ev-desc').textContent = ev.text;
    const btnBox = $('ev-btns');
    btnBox.innerHTML = '';

    ev.btns.forEach(b => {
      const el = document.createElement('button');
      el.className = 'btn btn-green';
      el.textContent = b.text;
      el.onclick = () => {
        m.classList.remove('active');
        b.act(this);
        this.ui.bars();
        this.ui.header();
        this.save();
      };
      btnBox.appendChild(el);
    });

    m.classList.add('active');
  }

  // ------------------------------------------------------
  // БОЕВАЯ СИСТЕМА
  // ------------------------------------------------------
  startCombat(loc) {
    const eKey = loc.enemies[Math.floor(Math.random() * loc.enemies.length)];
    const proto = DB.enemies[eKey];
    const e = Object.assign({}, proto, { hp: proto.hp, maxHp: proto.hp });

    this.combat = { enemy: e, loc: loc, round: 1 };
    $('c-e-icon').textContent = e.icon;
    $('c-e-name').textContent = e.name;
    $('c-e-hptxt').textContent = `${e.hp}/${e.maxHp}`;
    $('c-e-bar').style.width = '100%';
    $('c-round').textContent = 'Раунд 1';
    $('c-log').innerHTML = '';

    this.ui.combatLog(`⚠️ Нападение! [${e.name}] преграждает вам путь.`);
    this.audio.hit();
    $('combat-box').style.display = 'block';
    window.scrollTo({ top: 0, behavior: 'smooth' });
  }

  combatAction(act) {
    if (!this.combat) return;

    // MANDATORY NEEDS CHECK IN COMBAT: NO ATTACKING IF STARVING OR DEHYDRATED!
    if (!this.canAct(true)) return;

    const e = this.combat.enemy;

    // Player Turn
    let pDmg = 0;
    const critChance = 0.12 + (this.state.stats.agi * 0.015);

    if (act === 'attack') {
      const atkRange = this.getAtk();
      pDmg = Math.floor(Math.random() * (atkRange[1] - atkRange[0] + 1)) + atkRange[0];
      const crit = Math.random() < critChance;
      if (crit) pDmg = Math.floor(pDmg * 1.65);
      e.hp -= pDmg;
      this.audio.hit();
      this.ui.combatLog(`Вы нанесли ${crit ? '💥 КРИТИЧЕСКИЙ ' : ''}${pDmg} урона по ${e.name}!`);
    } else if (act === 'heavy') {
      if (this.state.energy < 4) { this.ui.toast("Недостаточно энергии!", "red"); return; }
      this.state.energy -= 4;
      const atkRange = this.getAtk();
      pDmg = Math.floor((atkRange[1] * 1.55) + Math.random() * 12);
      e.hp -= pDmg;
      this.audio.hit();
      this.ui.combatLog(`💥 Мощный выпад: ${pDmg} урона по ${e.name}!`);
    } else if (act === 'heal') {
      const med = this.inv.has("med_kit") ? "med_kit" : (this.inv.has("med_bandage") ? "med_bandage" : null);
      if (!med) { this.ui.toast("Нет бинтов или аптечек!", "red"); return; }
      this.inv.use(med);
      this.ui.combatLog(`Вы применили ${DB.items[med].name} в бою!`);
    } else if (act === 'flee') {
      if (Math.random() < 0.55) {
        this.ui.combatLog("💨 Вы успешно оторвались от преследования!");
        this.endCombat(false);
        return;
      } else {
        this.ui.combatLog(`❌ Не удалось сбежать! ${e.name} преградил путь!`);
      }
    }

    // Slightly burn thirst during strenuous combat
    this.state.thirst = Math.max(0, Number((this.state.thirst - 1.2).toFixed(1)));
    this.ui.bars();

    // Check Enemy Death
    if (e.hp <= 0) {
      this.combatVictory();
      return;
    }

    if (!this.combat) return;

    // Enemy Turn
    setTimeout(() => {
      if (!this.combat) return;
      const eDmgRange = e.dmg;
      let rawDmg = Math.floor(Math.random() * (eDmgRange[1] - eDmgRange[0] + 1)) + eDmgRange[0];
      const playerDef = this.getDef();
      let finalDmg = Math.max(2, rawDmg - Math.floor(playerDef * 0.35));

      const dodgeChance = 0.08 + (this.state.stats.agi * 0.01);
      if (Math.random() < dodgeChance) {
        this.ui.combatLog(`⚡ Вы ловко увернулись от удара ${e.name}!`);
      } else {
        this.state.hp = Math.max(0, this.state.hp - finalDmg);
        this.audio.hit();
        this.ui.combatLog(`${e.name} атаковал на ${finalDmg} урона!`);
      }

      this.ui.bars();
      if (this.state.hp <= 0) {
        this.combatDefeat();
      } else {
        this.combat.round++;
        $('c-round').textContent = `Раунд ${this.combat.round}`;
      }
    }, 180);

    // Update Enemy Bar
    $('c-e-hptxt').textContent = `${Math.max(0, e.hp)}/${e.maxHp}`;
    $('c-e-bar').style.width = `${Math.max(0, (e.hp / e.maxHp) * 100)}%`;
  }

  combatVictory() {
    const e = this.combat.enemy;
    this.state.tracker.kills++;
    this.state.exp += e.exp;

    // Perception stat grants bonus rubles
    const perCashMult = 1 + (this.state.stats.per * 0.04);
    let rubReward = Math.floor((Math.random() * (e.rubles[1] - e.rubles[0] + 1) + e.rubles[0]) * perCashMult);
    this.state.rubles += rubReward;

    let dropTxt = "";
    if (Math.random() < e.lootChance && e.drop) {
      this.inv.add(e.drop, 1);
      dropTxt = `, Трофей: [${DB.items[e.drop].rarityName}] ${DB.items[e.drop].name}`;
    }

    this.audio.loot();
    this.ui.toast(`🏆 Победа над ${e.name}! +${e.exp} EXP, +${rubReward}₽${dropTxt}`, "gold");
    this.log.add(`Уничтожен ${e.name}. Добыча: ${rubReward}₽, ${e.exp} EXP.`);
    this.checkLvl();
    this.checkQ();
    this.checkAch();
    this.endCombat(true);
  }

  combatDefeat() {
    this.audio.hit();
    this.ui.toast("☠️ Вы потеряли сознание в бою! Сталкеры оттащили вас в лагерь.", "red");
    this.log.add("Поражение в схватке. Потеряна часть припасов.");
    this.state.hp = 25;
    this.state.energy = 25;
    this.state.rubles = Math.max(0, Math.floor(this.state.rubles * 0.8));
    this.endCombat(false);
  }

  endCombat(won) {
    this.combat = null;
    $('combat-box').style.display = 'none';
    this.ui.bars();
    this.ui.header();
    this.save();
  }

  // ------------------------------------------------------
  // КВЕСТЫ И ДОСТИЖЕНИЯ
  // ------------------------------------------------------
  checkQ() {
    DB.quests.forEach(q => {
      let s = this.state.quests[q.id];
      if (!s) {
        s = { completed: false, claimed: false };
        this.state.quests[q.id] = s;
      }
      if (s.completed) return;

      let val = 0;
      if (q.type === 'kills') val = this.state.tracker.kills;
      else if (q.type === 'searches') val = this.state.tracker.searches;
      else if (q.type === 'equip_gun') {
        const w = DB.items[this.state.equipment.weapon];
        val = (w && w.icon !== "🗡️") ? 1 : 0;
      } else if (q.type === 'equip_art') {
        val = this.state.equipment.artifact ? 1 : 0;
      } else if (q.type === 'level') {
        val = this.state.level;
      } else if (this.state.tracker[q.type] !== undefined) {
        val = this.state.tracker[q.type];
      }

      if (val >= q.target) {
        s.completed = true;
        this.audio.jackpot();
        this.ui.toast(`📜 ЗАДАНИЕ ВЫПОЛНЕНО: «${q.title}»! Заберите награду.`, "gold");
        this.log.add(`Выполнено задание: «${q.title}». Заберите награду в КПК!`);
      }
    });
    this.updateRewardBadges();
    this.ui.quests();
  }

  checkAch() {
    DB.achievements.forEach(a => {
      let s = this.state.achievements[a.id];
      if (!s || typeof s !== 'object') {
        s = { unlocked: false, claimed: false };
        this.state.achievements[a.id] = s;
      }
      if (!s.unlocked && a.check(this.state)) {
        s.unlocked = true;
        this.audio.jackpot();
        this.ui.toast(`🏆 ДОСТИЖЕНИЕ: «${a.title}»! Награда готова к получению!`, "gold");
        this.log.add(`Разблокировано достижение «${a.title}»!`);
      }
    });
    this.updateRewardBadges();
    this.ui.achievements();
  }

  claimQ(qid) {
    const q = DB.quests.find(x => x.id === qid);
    if (!q) return;
    const s = this.state.quests[qid];
    if (!s || !s.completed || s.claimed) {
      if (s && s.claimed) this.ui.toast("Награда за это задание уже получена!");
      return;
    }
    s.claimed = true;
    this.state.rubles += q.reward.rubles;
    this.state.exp += q.reward.exp;
    if (q.reward.gram) {
      this.state.gramcoin = Number((this.state.gramcoin + q.reward.gram).toFixed(2));
      this.state.tracker.gramTotal = Number((this.state.tracker.gramTotal + q.reward.gram).toFixed(2));
    }
    if (q.reward.item) this.inv.add(q.reward.item, q.reward.count || 1);

    this.audio.loot();
    this.ui.toast(`🎁 Награда за задание «${q.title}» получена!`, "gold");
    this.log.add(`Получена награда за задание «${q.title}».`);
    this.checkLvl();
    this.updateRewardBadges();
    this.ui.quests();
    this.ui.header();
    this.save();
  }

  // ROBUST CLAIMING WITH FALLBACK
  claimAch(aid) {
    const a = DB.achievements.find(x => x.id === aid);
    if (!a) return;
    let s = this.state.achievements[aid];
    if (!s || typeof s !== 'object') {
      s = { unlocked: false, claimed: false };
      this.state.achievements[aid] = s;
    }
    const meetsCondition = a.check(this.state);
    if ((!s.unlocked && !meetsCondition) || s.claimed) {
      if (s.claimed) this.ui.toast("Награда за это достижение уже получена!");
      return;
    }
    s.unlocked = true;
    s.claimed = true;
    if (a.reward) {
      if (a.reward.rubles) this.state.rubles += a.reward.rubles;
      if (a.reward.exp) this.state.exp += a.reward.exp;
      if (a.reward.gram) {
        this.state.gramcoin = Number((this.state.gramcoin + a.reward.gram).toFixed(2));
        this.state.tracker.gramTotal = Number((this.state.tracker.gramTotal + a.reward.gram).toFixed(2));
      }
      if (a.reward.item) this.inv.add(a.reward.item, a.reward.count || 1);
    }
    this.audio.loot();
    this.ui.toast(`🏆 Награда за достижение «${a.title}» получена!`, "gold");
    this.log.add(`Получена награда за достижение «${a.title}».`);
    this.checkLvl();
    this.updateRewardBadges();
    this.ui.achievements();
    this.ui.header();
    this.save();
  }

  claimAllRewards() {
    let count = 0;
    DB.quests.forEach(q => {
      const s = this.state.quests[q.id];
      if (s && s.completed && !s.claimed) {
        s.claimed = true;
        this.state.rubles += q.reward.rubles;
        this.state.exp += q.reward.exp;
        if (q.reward.gram) {
          this.state.gramcoin = Number((this.state.gramcoin + q.reward.gram).toFixed(2));
          this.state.tracker.gramTotal = Number((this.state.tracker.gramTotal + q.reward.gram).toFixed(2));
        }
        if (q.reward.item) this.inv.add(q.reward.item, q.reward.count || 1);
        count++;
      }
    });
    DB.achievements.forEach(a => {
      let s = this.state.achievements[a.id];
      if (!s || typeof s !== 'object') {
        s = { unlocked: false, claimed: false };
        this.state.achievements[a.id] = s;
      }
      const meetsCondition = a.check(this.state);
      if ((s.unlocked || meetsCondition) && !s.claimed) {
        s.unlocked = true;
        s.claimed = true;
        if (a.reward) {
          if (a.reward.rubles) this.state.rubles += a.reward.rubles;
          if (a.reward.exp) this.state.exp += a.reward.exp;
          if (a.reward.gram) {
            this.state.gramcoin = Number((this.state.gramcoin + a.reward.gram).toFixed(2));
            this.state.tracker.gramTotal = Number((this.state.tracker.gramTotal + a.reward.gram).toFixed(2));
          }
          if (a.reward.item) this.inv.add(a.reward.item, a.reward.count || 1);
        }
        count++;
      }
    });

    if (count > 0) {
      this.audio.jackpot();
      this.ui.toast(`🎁 УСПЕШНО ЗАБРАНО НАГРАД: ${count}!`, "gold");
      this.log.add(`Забраны все готовые награды (всего ${count} шт.).`);
      this.checkLvl();
      this.updateRewardBadges();
      this.ui.quests();
      this.ui.achievements();
      this.ui.header();
      this.save();
    } else {
      this.ui.toast("Нет наград, готовых к получению!");
    }
  }

  updateRewardBadges() {
    let unclaimed = 0;
    DB.quests.forEach(q => {
      const s = this.state.quests[q.id];
      if (s && s.completed && !s.claimed) unclaimed++;
    });
    DB.achievements.forEach(a => {
      const s = this.state.achievements[a.id];
      const meets = a.check(this.state);
      if (((s && s.unlocked) || meets) && !(s && s.claimed)) unclaimed++;
    });

    const badge = $('nav-qst-badge');
    if (badge) {
      if (unclaimed > 0) {
        badge.textContent = unclaimed;
        badge.style.display = 'block';
      } else {
        badge.style.display = 'none';
      }
    }

    const banner = $('claim-all-banner');
    const bannerCnt = $('claim-all-cnt');
    if (banner && bannerCnt) {
      if (unclaimed > 0) {
        banner.style.display = 'block';
        bannerCnt.textContent = unclaimed;
      } else {
        banner.style.display = 'none';
      }
    }
  }
}

// МЕНЕДЖЕР ИНВЕНТАРЯ И ПОТРЕБНОСТЕЙ
// --------------------------------------------------------
class InvManager {
  constructor(g) { this.g = g; }

  has(id, cnt = 1) {
    const r = this.g.state.inventory.find(i => i.id === id);
    return r && r.count >= cnt;
  }

  add(id, cnt = 1) {
    const r = this.g.state.inventory.find(i => i.id === id);
    if (r) r.count += cnt;
    else this.g.state.inventory.push({ id, count: cnt });
    this.g.ui.inventory();
  }

  remove(id, cnt = 1) {
    const idx = this.g.state.inventory.findIndex(i => i.id === id);
    if (idx !== -1) {
      this.g.state.inventory[idx].count -= cnt;
      if (this.g.state.inventory[idx].count <= 0) this.g.state.inventory.splice(idx, 1);
    }
    this.g.ui.inventory();
  }

  equip(id) {
    const it = DB.items[id]; if (!it) return;
    const cur = this.g.state.equipment[it.type];
    if (cur) this.add(cur, 1);
    this.remove(id, 1);
    this.g.state.equipment[it.type] = id;
    this.g.recalc();
    this.g.audio.play(600);
    this.g.ui.toast(`Экипировано: [${it.rarityName}] ${it.name}`);
    this.g.ui.gear();
    this.g.ui.inventory();
    this.g.ui.bars();
    this.g.checkQ();
    this.g.checkAch();
    this.g.save();
  }

  use(id) {
    const it = DB.items[id]; if (!it) return;
    if (['weapon', 'armor', 'artifact'].includes(it.type)) {
      this.equip(id);
      return;
    }

    let used = false;

    // Food & Drinks consumption
    if (it.type === 'food') {
      if (it.hunger) {
        this.g.state.hunger = Math.min(100, this.g.state.hunger + it.hunger);
        used = true;
      }
      if (it.thirst) {
        this.g.state.thirst = Math.min(100, this.g.state.thirst + it.thirst);
        used = true;
      }
      if (it.heal && this.g.state.hp < this.g.state.maxHp) {
        this.g.state.hp = Math.min(this.g.state.maxHp, this.g.state.hp + it.heal);
        used = true;
      }
      if (it.radHeal && this.g.state.radiation > 0) {
        this.g.state.radiation = Math.max(0, this.g.state.radiation - it.radHeal);
        used = true;
      }
      if (it.energyBonus && this.g.state.energy < this.g.state.maxEnergy) {
        this.g.state.energy = Math.min(this.g.state.maxEnergy, this.g.state.energy + it.energyBonus);
        used = true;
      }
    } else if (it.type === 'med') {
      if (it.heal && this.g.state.hp < this.g.state.maxHp) {
        this.g.state.hp = Math.min(this.g.state.maxHp, this.g.state.hp + it.heal);
        used = true;
      }
      if (it.radHeal && this.g.state.radiation > 0) {
        this.g.state.radiation = Math.max(0, this.g.state.radiation - it.radHeal);
        used = true;
      }
      if (it.energyBonus && this.g.state.energy < this.g.state.maxEnergy) {
        this.g.state.energy = Math.min(this.g.state.maxEnergy, this.g.state.energy + it.energyBonus);
        used = true;
      }
      if (it.hunger) {
        this.g.state.hunger = Math.min(100, this.g.state.hunger + it.hunger);
        used = true;
      }
      if (it.thirst) {
        this.g.state.thirst = Math.min(100, this.g.state.thirst + it.thirst);
        used = true;
      }
    }

    if (used) {
      this.remove(id, 1);
      this.g.audio.play(700);
      this.g.ui.toast(`Употреблено: ${it.name}`);
      this.g.ui.bars();
      this.g.ui.updateNeedsBanner();
      this.g.save();
    } else {
      this.g.ui.toast("Организм полон сил, сейчас это не требуется!");
    }
  }
}

class GramManager {
  constructor(g) { this.g = g; }

  roll(chance) {
    if (Math.random() < chance) {
      const drop = Number((Math.random() * 2.9 + 0.1).toFixed(2));
      this.g.state.gramcoin = Number((this.g.state.gramcoin + drop).toFixed(2));
      this.g.state.tracker.gramTotal = Number((this.g.state.tracker.gramTotal + drop).toFixed(2));

      this.g.audio.jackpot();
      $('m-j-amt').textContent = `+${drop.toFixed(2)} Gramcoin`;
      $('m-jackpot').classList.add('active');

      this.g.log.add(`💎 РЕДЧАЙШАЯ НАХОДКА! Получено +${drop.toFixed(2)} Gramcoin!`, 'blue');
      this.g.ui.toast(`💎 АНОМАЛЬНЫЙ ДЖЕКПОТ! +${drop.toFixed(2)} Gramcoin!`, 'blue');
      this.g.ui.header();
      this.g.ui.profile();
      this.g.checkAch();
      this.g.save();
      return true;
    }
    return false;
  }
}

class TraderManager {
  constructor(g) { this.g = g; }

  buy(id) {
    const it = DB.items[id]; if (!it) return;
    if (this.g.state.rubles < it.price) {
      this.g.ui.toast("Недостаточно денег!", "red");
      return;
    }
    this.g.state.rubles -= it.price;
    this.g.inv.add(id, 1);
    this.g.audio.play(500);
    this.g.ui.toast(`Куплено: ${it.name} (-${it.price}₽)`);
    this.g.log.add(`Куплено у Сидоровича: ${it.name} за ${it.price}₽.`);
    this.g.ui.header();
    this.g.checkAch();
    this.g.save();
  }

  sell(id) {
    const it = DB.items[id]; if (!it) return;
    const sellPrice = Math.floor(it.price * 0.55);
    this.inv.remove(id, 1);
    this.g.state.rubles += sellPrice;
    this.g.audio.play(520);
    this.g.ui.toast(`Продано: ${it.name} (+${sellPrice}₽)`);
    this.g.log.add(`Продано: ${it.name} (+${sellPrice}₽).`);
    this.g.ui.header();
    this.g.ui.shop();
    this.g.checkAch();
    this.g.save();
  }
}

class LogManager {
  constructor(g) { this.g = g; }
  add(txt, cls = '') {
    const box = $('game-log'); if (!box) return;
    const el = document.createElement('div');
    el.className = `log-entry ${cls}`;
    const time = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });
    el.textContent = `[${time}] ${txt}`;
    box.prepend(el);
    while (box.children.length > 40) box.removeChild(box.lastChild);
  }
}

          <strong style="color:${isUnlocked ? (isClaimed ? 'var(--green)' : 'var(--gold)') : 'var(--muted)'}">
            ${isUnlocked ? '🏆' : '🔒'} ${a.title}
          </strong>
          ${actionHtml}
        </div>
        <div style="font-size:10px;color:var(--muted)">${a.desc}</div>
        <div style="font-size:10px;color:var(--gold);margin-top:3px;font-weight:700">Награда: ${rText}</div>
      `;
      box.appendChild(c);
    });
    $('ach-cnt').textContent = `${cnt}/${DB.achievements.length}`;
  }

  profile() {
    const s = this.g.state;
    $('p-name').textContent = s.name;
    $('p-rank').textContent = `${s.level} ур. (${s.title})`;
    $('p-exp').textContent = `${s.exp} / ${s.expNext} EXP`;
    $('p-kills').textContent = `${s.tracker.kills} мутантов`;
    $('p-searches').textContent = `${s.tracker.searches} вылазок`;
    $('p-gram-total').textContent = `${(s.tracker.gramTotal || s.gramcoin).toFixed(2)} G`;

    // Attributes Panel
    $('p-free-pts').textContent = s.statPoints;
    $('stat-str').textContent = s.stats.str;
    $('stat-agi').textContent = s.stats.agi;
    $('stat-end').textContent = s.stats.end;
    $('stat-per').textContent = s.stats.per;

    const hasPts = s.statPoints > 0;
    ['str', 'agi', 'end', 'per'].forEach(k => {
      const btn = $(`btn-plus-${k}`);
      if (btn) btn.disabled = !hasPts;
    });
  }

  log() {
    this.g.log.add("КПК активирован. Загружены карты 15 секторов и модуль метаболизма.");
  }

  combatLog(txt) {
    const box = $('c-log'); if (!box) return;
    const el = document.createElement('div');
    el.className = 'c-log-line';
    el.textContent = txt;
    box.prepend(el);
  }
